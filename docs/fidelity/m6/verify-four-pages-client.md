## 4. 复现命令（第三人可直接跑）

```powershell
# 1) 取原版 dump（只读基线，不入库改动）
adb pull /sdcard/a.xml $env:TEMP\orig.xml   # 或直接用 docs/fidelity/m3/*.xml

# 2) 逐节点量取（本报告同一算法：text/content-desc + class + bounds + clickable）
python - <<'PY'
import xml.etree.ElementTree as ET
for p in [r"docs/fidelity/m3/contact.xml", r"docs/fidelity/m3/account-setting.xml",
          r"docs/fidelity/m3/complaint.xml", r"docs/fidelity/m3/about.xml"]:
    print("==", p)
    for n in ET.parse(p).getroot().iter("node"):
        if n.get("bounds") in ("[0,0][0,0]", None): continue
        t = n.get("text") or n.get("content-desc") or ""
        print(n.get("class"), n.get("bounds"), "clickable=" + n.get("clickable"), t)
PY

# 3) 取色/圆角/蓝钮（Pillow）
python - <<'PY'
from PIL import Image
b=r"docs/fidelity/m3"
im=Image.open(b+r"\contact.png").convert("RGB")
print("card", [(x,im.getpixel((x,1200))) for x in (168,600,1058)])
print("blue", im.getpixel((600,1536)))
print("bg_account", Image.open(b+r"\account-setting.png").convert("RGB").getpixel((1200,1800)))
print("bg_about", Image.open(b+r"\about.png").convert("RGB").getpixel((20,600)))
PY
```
